Partie 4 — Interface de chat complète

Créer une interface de chat avec historique de conversation, en utilisant l'endpoint /api/chat d'Ollama.

40 min endpoint /api/chat historique de messages
4.1

/api/chat vs /api/generate

Vous venez d'utiliser /api/generate pour des prompts ponctuels. Pour une vraie conversation, Ollama expose /api/chat qui maintient le contexte grâce à un tableau de messages avec des rôles.

🔹 /api/generate

Reçoit un prompt (string)

Renvoie data.response

Usage : prompt unique, sans contexte

🔷 /api/chat

Reçoit des messages (array)

Renvoie data.message.content

Usage : conversation multi-tours avec historique

// Structure pour /api/chat - maintien du contexte
{
  "model": "mistral",
  "messages": [
    { "role": "system",    "content": "Tu es un assistant pédagogique MMI." },
    { "role": "user",      "content": "Qu'est-ce qu'une API REST ?" },
    { "role": "assistant", "content": "Une API REST est..." },
    { "role": "user",      "content": "Donne-moi un exemple concret en JS." }
  ],
  "stream": false
}

// La réponse est dans : data.message.content
💡 Différence clé
/api/generate reçoit un prompt (string) et renvoie data.response. /api/chat reçoit des messages (array) et renvoie data.message.content. C'est ce que Continue utilise en interne pour son chat.
4.2

Code : chat.html

Créez chat.html — le même format de chat que Continue, mais dans votre propre page :

<!DOCTYPE html>
<html lang="fr">
<head>
  <meta charset="UTF-8">
  <title>Chat Ollama</title>
  <style>
    * { box-sizing: border-box; margin: 0; padding: 0; }
    body { font-family: Arial, sans-serif; background: #f1f5f9;
           display: flex; flex-direction: column; height: 100vh; }
    header { background: #2563eb; color: white; padding: 14px 20px;
             display: flex; align-items: center; gap: 12px; }
    header h1 { font-size: 1.1rem; flex: 1; }
    select { padding: 6px 10px; border-radius: 6px; border: none; cursor: pointer; }
    button.sec { background: rgba(255,255,255,0.2); color: white;
                 border: 1px solid rgba(255,255,255,0.4); padding: 6px 12px;
                 border-radius: 6px; cursor: pointer; font-size: 0.85rem; }
    #messages { flex: 1; overflow-y: auto; padding: 16px;
                display: flex; flex-direction: column; gap: 10px; }
    .msg { max-width: 78%; padding: 12px 16px; border-radius: 12px;
           line-height: 1.5; font-size: 0.9rem; white-space: pre-wrap; }
    .user  { background: #2563eb; color: white; align-self: flex-end;
             border-bottom-right-radius: 3px; }
    .assistant { background: white; border: 1px solid #e2e8f0;
                 align-self: flex-start; border-bottom-left-radius: 3px; }
    .meta { font-size: 0.75rem; color: #94a3b8; margin-top: 4px; }
    .attente { opacity: 0.6; font-style: italic; }
    footer { background: white; border-top: 1px solid #e2e8f0;
             padding: 12px; display: flex; gap: 8px; }
    textarea { flex: 1; padding: 10px; border: 1px solid #cbd5e1;
               border-radius: 8px; resize: none; font-family: inherit;
               font-size: 0.9rem; }
    button#btn { padding: 10px 18px; background: #2563eb; color: white;
                 border: none; border-radius: 8px; cursor: pointer;
                 font-weight: bold; }
    button#btn:disabled { background: #94a3b8; }
  </style>
</head>
<body>
  <header>
    <h1>Chat Ollama</h1>
    <select id="modele">
      <option value="mistral">Mistral 7B</option>
      <option value="qwen2.5">Qwen2.5 7B</option>
      <option value="llama3.2">Llama 3.2 3B</option>
    </select>
    <button class="sec" onclick="reinit()">Nouvelle conversation</button>
  </header>

  <div id="messages"></div>

  <footer>
    <textarea id="saisie" rows="2"
      placeholder="Tapez votre message... (Entrée pour envoyer, Maj+Entrée pour saut de ligne)"
      onkeydown="if(event.key==='Enter'&&!event.shiftKey){event.preventDefault();envoyer()}"></textarea>
    <button id="btn" onclick="envoyer()">Envoyer</button>
  </footer>

  <script>
    const API = "http://localhost:11434/api/chat";
    let historique = [];

    function ajouterMsg(role, texte, modele = "") {
      const zone = document.getElementById("messages");
      const div = document.createElement("div");
      div.className = "msg " + role;
      div.textContent = texte;
      if (role === "assistant" && modele) {
        const meta = document.createElement("div");
        meta.className = "meta";
        meta.textContent = "⚡ " + modele;
        div.appendChild(meta);
      }
      zone.appendChild(div);
      zone.scrollTop = zone.scrollHeight;
      return div;
    }

    async function envoyer() {
      const saisie = document.getElementById("saisie");
      const btn = document.getElementById("btn");
      const texte = saisie.value.trim();
      if (!texte) return;
      const modele = document.getElementById("modele").value;
      saisie.value = "";
      btn.disabled = true;
      ajouterMsg("user", texte);
      historique.push({ role: "user", content: texte });
      const msgEnCours = ajouterMsg("assistant", "Rédaction en cours...");
      msgEnCours.classList.add("attente");
      try {
        const res = await fetch(API, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ model: modele, messages: historique, stream: false })
        });
        const data = await res.json();
        const rep = data.message.content;
        msgEnCours.textContent = rep;
        msgEnCours.classList.remove("attente");
        const meta = document.createElement("div");
        meta.className = "meta";
        meta.textContent = "⚡ " + modele;
        msgEnCours.appendChild(meta);
        historique.push({ role: "assistant", content: rep });
      } catch (err) {
        msgEnCours.textContent = "Erreur : " + err.message;
      } finally {
        btn.disabled = false;
        saisie.focus();
      }
    }

    function reinit() {
      historique = [];
      document.getElementById("messages").innerHTML = "";
    }
  </script>
</body>
</html>
4.3

Améliorations guidées

Implémentez au moins 2 de ces améliorations. Utilisez Continue pour vous aider !

Amélioration 1 Message système configurable

Dans /api/chat, le premier message peut avoir le rôle "system". C'est une instruction invisible pour l'utilisateur qui configure le comportement du modèle pour toute la conversation — un "briefing" donné au LLM avant qu'il réponde.


1. Ajouter un <input> en haut de la page :

<input id="systeme" type="text"
  placeholder="Ex : Tu es un assistant pédagogique MMI, réponds en français, de façon concise."
  style="width:100%; padding:8px; border:1px solid #cbd5e1; border-radius:6px; font-size:0.9rem;">

2. Dans la fonction envoyer(), injecter le message système en tête du tableau messages :

async function envoyer() {
  // ... récupération du texte, modèle, etc.

  const systeme = document.getElementById('systeme').value.trim();
  const messages = [];
  if (systeme) {
    messages.push({ role: 'system', content: systeme });
  }
  messages.push(...historique);

  const res = await fetch(API, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ model: modele, messages, stream: false })
  });
  // ...
}

3. Testez avec différents messages système :

  • "Tu es un expert en HTML/CSS. Réponds uniquement avec du code, sans explication."
  • "Tu es un professeur qui explique à des étudiants de BUT MMI. Sois pédagogique et utilise des analogies."
  • "Tu réponds uniquement en JSON structuré, sans texte libre."
💡 L'intérêt pédagogique
Changer ce seul champ transforme radicalement le comportement du modèle sans toucher au reste du code. C'est le principe fondamental du prompt engineering — une compétence clé pour intégrer des LLM dans vos projets.

Au choix Autres améliorations

  • Afficher le temps de réponse en millisecondes sous chaque message assistant
  • Ajouter un bouton Copier sur chaque réponse (navigator.clipboard.writeText)
  • Sauvegarder l'historique dans localStorage et le restaurer au rechargement
  • Implémenter le streaming : stream: true + lecture du ReadableStream token par token
💡 Astuce Continue
Sélectionnez la fonction envoyer() dans VS Code, ouvrez le chat Continue (Cmd+L) et demandez : "Modifie cette fonction pour afficher le temps de réponse en ms sous chaque message". Comparez Mistral vs Qwen2.5 sur cette tâche de code.
